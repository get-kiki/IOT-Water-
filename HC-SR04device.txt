   WiFi: จัดการด้วย WiFiManager (tzapu)

   โหมดใช้งานจริง (TEST_PORTAL ปิดอยู่)
   บอร์ดจะต่อ WiFi เดิมที่เคยตั้งไว้อัตโนมัติเมื่อบูต ถ้าต่อไม่ติดค่อยเปิดหน้า
   ตั้งค่าให้เอง ถ้าจะทดสอบเปลี่ยน WiFi ให้ไปเปิดบรรทัด #define TEST_PORTAL
   ด้านล่างใหม่ (เอา // ออก) แล้วอย่าลืมปิดกลับก่อนใช้งานจริงอีกครั้ง

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <WiFi.h>
#include <WiFiManager.h>          // ติดตั้ง "WiFiManager" by tzapu
#include <FirebaseESP32.h>
#include <EEPROM.h>

// ---------- โหมดทดสอบ ----------
// #define TEST_PORTAL             // <<< ปิดแล้ว (โหมดใช้งานจริง) เปิดใหม่ได้ถ้าต้องทดสอบ WiFi

// ---------- Firebase ----------
#define FIREBASE_HOST "water-tank-iot-8c113-default-rtdb.asia-southeast1.firebasedatabase.app"
#define FIREBASE_AUTH "qKEAYzn9dc31GnN38eAaHFtmUWcIxyilFFh43beN"

// ---------- WiFiManager ----------
#define AP_NAME            "WaterTank-Setup"   // ชื่อ WiFi ที่บอร์ดปล่อยตอนตั้งค่า
#define AP_PASSWORD        "12345678"          // รหัสเข้า AP (ต้อง >= 8 ตัว)
#define PORTAL_TIMEOUT_SEC 180                 // ไม่มีใครตั้งค่าใน 3 นาที -> ทำงานต่อ
#define CONNECT_TIMEOUT_SEC 20                 // รอต่อ WiFi เดิม 20 วิ

// ---------- OLED ----------
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// ---------- ขา ----------
#define TRIG_PIN   25
#define ECHO_PIN   18
#define BUTTON_PIN 0     // ปุ่ม BOOT: กดค้าง 3 วิ = ล้างค่า WiFi

#define EEPROM_SIZE 8    // float 2 ตัว x 4 bytes

FirebaseData firebaseData;
FirebaseConfig config;
FirebaseAuth auth;

bool firebaseReady = false;

float TANK_DEPTH    = 114.0;
float TANK_CAPACITY = 400.0;

float previousWaterHeight = 0.0;
float smoothPercent       = 0.0;
int   rawWaterPercent     = 0;
float currentWaterHeight  = 0.0;

const float SAFE_ZONE = 2.5;

unsigned long lastSensorReadTime    = 0;
unsigned long lastFlowCheckTime     = 0;
unsigned long lastFirebaseCheckTime = 0;
unsigned long lastWifiCheckTime     = 0;

const unsigned long SENSOR_READ_INTERVAL    = 1000;
const unsigned long FLOW_CHECK_INTERVAL     = 5000;
const unsigned long FIREBASE_CHECK_INTERVAL = 3000;
const unsigned long WIFI_CHECK_INTERVAL     = 10000;

// ปุ่มกดค้างเพื่อล้าง WiFi
const unsigned long RESET_HOLD_MS = 3000;      // อยากให้กดนานขึ้นก็เพิ่มเลขนี้
unsigned long buttonHoldStart = 0;
bool buttonWasDown = false;

String flowStatus = "NO FLOW";

// -------------------------------------------------------
// แสดงข้อความ 3 บรรทัดบน OLED
void showMessage(String l1, String l2, String l3) {
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 10); display.print(l1);
  display.setCursor(0, 26); display.print(l2);
  display.setCursor(0, 42); display.print(l3);
  display.display();
}

// -------------------------------------------------------
// WiFiManager เรียกอัตโนมัติเมื่อเข้าโหมดตั้งค่า
void configModeCallback(WiFiManager *myWiFiManager) {
  Serial.println("[WiFi] เข้าโหมดตั้งค่า - ต่อมือถือเข้า " AP_NAME);
  showMessage("SETUP WiFi", AP_NAME, "IP 192.168.4.1");
}

// -------------------------------------------------------
// เริ่ม Firebase (เรียกซ้ำได้ ทำงานจริงครั้งเดียว)
void initFirebase() {
  if (firebaseReady) return;
  config.host = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
  firebaseReady = true;
  Serial.println("=================================");
  Serial.println("บอร์ดออนไลน์สำเร็จ!");
  Serial.print("SSID: "); Serial.println(WiFi.SSID());
  Serial.print("IP  : "); Serial.println(WiFi.localIP());
  Serial.println("=================================");
}

// -------------------------------------------------------
// กดปุ่ม BOOT ค้าง 3 วิ = ลืมรหัส WiFi แล้วรีสตาร์ทเข้าโหมดตั้งค่า
void handleWiFiResetButton() {
  if (digitalRead(BUTTON_PIN) == LOW) {
    if (!buttonWasDown) {
      buttonWasDown   = true;
      buttonHoldStart = millis();
    } else if (millis() - buttonHoldStart >= RESET_HOLD_MS) {
      Serial.println("[WiFi] ล้างค่า WiFi ที่บันทึกไว้...");
      showMessage("ERASE WiFi", "Restarting...", "");
      WiFiManager wm;
      wm.resetSettings();
      delay(1000);
      ESP.restart();
    }
  } else {
    buttonWasDown = false;
  }
}

// -------------------------------------------------------
float readUltrasonic() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return 0;
  return duration * 0.034 / 2;
}

// -------------------------------------------------------
void drawWaterTank(int x, int y, int w, int h, int percent) {
  display.drawRoundRect(x, y, w, h, 4, SSD1306_WHITE);
  display.drawRect(x + 2, y - 2, w - 4, 2, SSD1306_WHITE);
  display.drawRect(x - 2, y + 4, 2, 2, SSD1306_WHITE);
  display.drawRect(x + w, y + 4, 2, 2, SSD1306_WHITE);
  int maxWaterHeight = h - 6;
  int fillH = maxWaterHeight * percent / 100;
  if (fillH > 0) {
    display.fillRect(x + 2, y + (h - 2) - fillH, w - 4, fillH, SSD1306_WHITE);
  }
}

// -------------------------------------------------------
void setup() {
  Serial.begin(115200);
  EEPROM.begin(EEPROM_SIZE);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) { for (;;); }
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);

  // ---- โหลดค่าที่บันทึกไว้จาก EEPROM ----
  float savedDepth, savedCapacity;
  EEPROM.get(0, savedDepth);
  EEPROM.get(4, savedCapacity);

  if (!isnan(savedDepth) && savedDepth > 10.0 && savedDepth < 500.0) {
    TANK_DEPTH = savedDepth;
    Serial.printf("[EEPROM] โหลด TANK_DEPTH = %.1f ซม.\n", TANK_DEPTH);
  }
  if (!isnan(savedCapacity) && savedCapacity > 10.0 && savedCapacity < 50000.0) {
    TANK_CAPACITY = savedCapacity;
    Serial.printf("[EEPROM] โหลด TANK_CAPACITY = %.0f ลิตร\n", TANK_CAPACITY);
  }

  // ---- เชื่อม WiFi ด้วย WiFiManager ----
  WiFiManager wm;
  wm.setAPCallback(configModeCallback);
  wm.setConfigPortalTimeout(PORTAL_TIMEOUT_SEC);
  wm.setConnectTimeout(CONNECT_TIMEOUT_SEC);
  wm.setTitle("Water Tank IoT");

  bool ok;

#ifdef TEST_PORTAL
  // โหมดทดสอบ: เปิดหน้าตั้งค่าทันทีทุกครั้ง ไม่สนรหัสเก่า
  // (ไม่ลบรหัสเก่าทิ้ง - ถ้าเปลี่ยนใจ กด Exit ในหน้าเว็บได้)
  Serial.println("[TEST] TEST_PORTAL เปิดอยู่ - บังคับเข้าหน้าตั้งค่า");
  showMessage("TEST MODE", "forcing portal", "");
  delay(800);
  ok = wm.startConfigPortal(AP_NAME, AP_PASSWORD);
#else
  // โหมดใช้งานจริง: มีรหัสเก่า -> ต่อเลย / ไม่มีหรือต่อไม่ติด -> เปิด portal
  showMessage("Connecting WiFi", "please wait...", "");
  ok = wm.autoConnect(AP_NAME, AP_PASSWORD);
#endif

  if (ok) {
    showMessage("WiFi CONNECTED", WiFi.SSID(), WiFi.localIP().toString());
    initFirebase();
  } else {
    Serial.println("[WiFi] ต่อไม่สำเร็จ - ทำงานแบบ offline");
    showMessage("WiFi FAILED", "run offline", "hold BTN 3s");
  }

  WiFi.setAutoReconnect(true);
  delay(1500);
}

// -------------------------------------------------------
void loop() {

  // กดปุ่ม BOOT ค้าง 3 วิ = เปลี่ยน WiFi ใหม่
  handleWiFiResetButton();

  // ---- เฝ้าดู WiFi ทุก 10 วินาที ----
  if (millis() - lastWifiCheckTime >= WIFI_CHECK_INTERVAL) {
    lastWifiCheckTime = millis();
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println("[WiFi] หลุด - กำลังต่อใหม่...");
      WiFi.reconnect();
    } else {
      initFirebase();   // เผื่อบูตมาแบบ offline แล้วเพิ่งต่อติดทีหลัง
    }
  }

  bool online = (WiFi.status() == WL_CONNECTED) && firebaseReady;

  // ---- อ่าน tank_depth / tank_capacity จาก Firebase ----
  if (online) {
    if (millis() - lastFirebaseCheckTime >= FIREBASE_CHECK_INTERVAL) {

      bool depthOk   = Firebase.getFloat(firebaseData, "/water_system/setting/tank_depth");
      float newDepth = depthOk ? firebaseData.floatData() : -1;

      bool capOk     = Firebase.getFloat(firebaseData, "/water_system/setting/tank_capacity");
      float newCap   = capOk ? firebaseData.floatData() : -1;

      if (depthOk && capOk && newDepth > 10.0 && newCap > 10.0) {
        if (newDepth != TANK_DEPTH || newCap != TANK_CAPACITY) {
          TANK_DEPTH    = newDepth;
          TANK_CAPACITY = newCap;
          EEPROM.put(0, TANK_DEPTH);
          EEPROM.put(4, TANK_CAPACITY);
          EEPROM.commit();
          Serial.printf("[OK] รับค่าจากแอป: %.0f ลิตร / %.1f ซม.\n", TANK_CAPACITY, TANK_DEPTH);
        }
      } else {
        if (!depthOk) Serial.println("อ่าน tank_depth ล้มเหลว: " + firebaseData.errorReason());
        if (!capOk)   Serial.println("อ่าน tank_capacity ล้มเหลว: " + firebaseData.errorReason());
      }

      lastFirebaseCheckTime = millis();
    }
  }

  // ---- อ่านเซนเซอร์ระดับน้ำทุก 1 วินาที ----
  if (millis() - lastSensorReadTime >= SENSOR_READ_INTERVAL) {
    float distanceCm = readUltrasonic();
    lastSensorReadTime = millis();

    if (distanceCm <= 0) {
      // อ่านค่าไม่สำเร็จ (มีอะไรมาบังใกล้เกินไป/ไกลเกินไป/echo timeout)
      // ข้ามรอบนี้ไปเลย คง currentWaterHeight/rawWaterPercent ค่าล่าสุดที่เชื่อถือได้ไว้ก่อน
      // กันไม่ให้ตีความผิดเป็นถังเต็ม 100% ทุกครั้งที่เซนเซอร์อ่านพลาด
      Serial.println("[Sensor] อ่านค่า HC-SR04 ไม่สำเร็จ ข้ามรอบนี้");
    } else {
      float effectiveDistance = distanceCm - SAFE_ZONE;
      if (effectiveDistance < 0) effectiveDistance = 0;

      currentWaterHeight = TANK_DEPTH - effectiveDistance;
      if (currentWaterHeight < 0)          currentWaterHeight = 0;
      if (currentWaterHeight > TANK_DEPTH) currentWaterHeight = TANK_DEPTH;

      rawWaterPercent = (int)((currentWaterHeight / TANK_DEPTH) * 100.0);
    }

    if (online) {
      int currentLiters = (int)((rawWaterPercent / 100.0) * TANK_CAPACITY);
      Firebase.setInt(firebaseData, "/water_system/monitor/water_level_liters", currentLiters);
      // 🚀 ไม่เขียนค่า flow_rate ปลอมขึ้น Firebase อีกต่อไป —
      // flow_rate_min เป็นหน้าที่ของบอร์ด flow sensor ขาเข้าเท่านั้น (inFlowSensorDevice.txt)
      // (เดิมใช้ flowStatus ของบอร์ดนี้ปลอมค่าไว้ ทำให้ "สถานะการไหล" ในแอปขยับ
      //  ทุกครั้งที่มีอะไรมาบัง/ขยับใกล้ตัวเซนเซอร์ ทั้งที่ไม่มีน้ำไหลจริง)
    }
  }

  // ---- ตรวจการไหลของน้ำทุก 5 วินาที ----
  if (millis() - lastFlowCheckTime >= FLOW_CHECK_INTERVAL) {
    flowStatus = (currentWaterHeight > previousWaterHeight + 0.4) ? "FLOW" : "NO FLOW";
    previousWaterHeight = currentWaterHeight;
    lastFlowCheckTime   = millis();
  }

  // ---- Smooth animation ----
  if (smoothPercent < rawWaterPercent) {
    smoothPercent += 0.5;
    if (smoothPercent > rawWaterPercent) smoothPercent = rawWaterPercent;
  } else if (smoothPercent > rawWaterPercent) {
    smoothPercent -= 0.5;
    if (smoothPercent < rawWaterPercent) smoothPercent = rawWaterPercent;
  }

  int displayPercent = (int)smoothPercent;
  int waterLiters    = (int)((displayPercent / 100.0) * TANK_CAPACITY);

  // ---- วาดหน้าจอ OLED ----
  display.clearDisplay();

  display.setTextSize(2);
  display.setCursor(4, 18);
  display.print(displayPercent);
  display.setTextSize(1);
  display.print("%");

  drawWaterTank(52, 6, 24, 42, displayPercent);

  display.setTextSize(1);
  display.setCursor(46, 50);
  display.print("H:");
  display.print((int)TANK_DEPTH);

  display.setCursor(110, 50);
  display.print(WiFi.status() == WL_CONNECTED ? "[ON]" : "[OFF]");

  if (flowStatus == "FLOW") {
    display.setCursor(88, 18); display.print("FLOW");
    display.setCursor(88, 28); display.print(">>>");
  } else {
    display.setCursor(84, 22); display.print("NO FLOW");
  }

  display.setTextSize(1);
  display.setCursor(4, 56);
  display.print(waterLiters);
  display.print(" / ");
  display.print((int)TANK_CAPACITY);
  display.print(" L");

  display.display();
  delay(50);
}
