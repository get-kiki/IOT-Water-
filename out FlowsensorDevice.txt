// YF-S201: ความถี่พัลส์ (Hz) = 7.5 * อัตราการไหล (L/min)  =>  450 พัลส์ ต่อ 1 ลิตร

// 🚀 ตัวนับพัลส์จาก Flow Sensor
volatile unsigned long flowPulseCount = 0;

float  currentFlowRate = 0.0;
String flowStatus      = "NO FLOW";

float  dailyLiters     = 0.0;
String currentDateKey  = "";

float  hourlyLiters    = 0.0;
int    currentHour     = -1;

unsigned long lastFlowCheckTime   = 0;
unsigned long lastPrefsSaveTime   = 0;
unsigned long lastHistoryPushTime = 0;
unsigned long lastWifiCheckTime   = 0;

const unsigned long FLOW_CHECK_INTERVAL   = 1000;
const unsigned long PREFS_SAVE_INTERVAL   = 30000;
const unsigned long HISTORY_PUSH_INTERVAL = 2000;
const unsigned long WIFI_CHECK_INTERVAL   = 10000;  // เฝ้าดู WiFi ทุก 10 วินาที

void IRAM_ATTR onFlowPulse() {
  flowPulseCount++;
}

// -------------------------------------------------------
String getTodayDateKey() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo, 2000)) {
    return "";
  }
  char buf[11];
  snprintf(buf, sizeof(buf), "%04d-%02d-%02d",
           timeinfo.tm_year + 1900, timeinfo.tm_mon + 1, timeinfo.tm_mday);
  return String(buf);
}

int getCurrentHour() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo, 2000)) {
    return -1;
  }
  return timeinfo.tm_hour;
}

String hourKeyStr(int hour) {
  char buf[3];
  snprintf(buf, sizeof(buf), "%02d", hour);
  return String(buf);
}

// -------------------------------------------------------
void loadTodayFromStorage() {
  String savedDate       = prefs.getString("date", "");
  float  savedLiters     = prefs.getFloat("liters", 0.0);
  int    savedHour       = prefs.getInt("hour", -1);
  float  savedHourLiters = prefs.getFloat("hliters", 0.0);

  String today   = getTodayDateKey();
  int    nowHour = getCurrentHour();

  if (today != "" && today == savedDate) {
    currentDateKey = savedDate;
    dailyLiters    = savedLiters;
    Serial.printf("[Storage] กู้ยอดวันนี้คืน: %.2f ลิตร (%s)\n", dailyLiters, currentDateKey.c_str());
  } else if (today != "") {
    currentDateKey = today;
    dailyLiters    = 0.0;
  }

  if (nowHour != -1 && today == savedDate && nowHour == savedHour) {
    currentHour  = savedHour;
    hourlyLiters = savedHourLiters;
    Serial.printf("[Storage] กู้ยอดชั่วโมงคืน: %.2f ลิตร (%02d:00)\n", hourlyLiters, currentHour);
  } else if (nowHour != -1) {
    currentHour  = nowHour;
    hourlyLiters = 0.0;
  }
}

void saveTodayToStorage() {
  prefs.putString("date", currentDateKey);
  prefs.putFloat("liters", dailyLiters);
  prefs.putInt("hour", currentHour);
  prefs.putFloat("hliters", hourlyLiters);
}

// แปลงรหัสสถานะ WiFi เป็นข้อความไทย ชี้สาเหตุได้ตรง ๆ
String wifiStatusText(wl_status_t s) {
  switch (s) {
    case WL_IDLE_STATUS:     return "IDLE (กำลังเริ่ม)";
    case WL_NO_SSID_AVAIL:   return "NO_SSID_AVAIL -> หาชื่อ WiFi ที่จำไว้ไม่เจอ (เราเตอร์ปิด? ชื่อเปลี่ยน? เป็น 5GHz?)";
    case WL_SCAN_COMPLETED:  return "SCAN_COMPLETED";
    case WL_CONNECTED:       return "CONNECTED (ต่อสำเร็จ)";
    case WL_CONNECT_FAILED:  return "CONNECT_FAILED -> เจอชื่อแล้วแต่เข้าไม่ได้ (รหัสผิด)";
    case WL_CONNECTION_LOST: return "CONNECTION_LOST -> หลุดกลางทาง";
    case WL_DISCONNECTED:    return "DISCONNECTED -> ยังไม่ได้ต่อ";
    default:                 return "UNKNOWN(" + String((int)s) + ")";
  }
}

// พิมพ์สาเหตุที่น่าจะต่อ WiFi ไม่ได้ (เรียกก่อน autoConnect)
void diagnoseWiFi() {
  WiFi.mode(WIFI_STA);
  delay(100);

  // ---- 1) บอร์ดจำ WiFi อะไรไว้ ----
  wifi_config_t conf;
  String saved = "";
  if (esp_wifi_get_config(WIFI_IF_STA, &conf) == ESP_OK) {
    saved = String((const char*)conf.sta.ssid);
  }
  if (saved.length() == 0) {
    Serial.println("[วินิจฉัย] บอร์ดยังไม่เคยจำ WiFi ไว้ -> จะเปิดหน้าตั้งค่าทันที");
  } else {
    Serial.println("[วินิจฉัย] WiFi ที่บอร์ดจำไว้ = \"" + saved + "\"");
  }

  // ---- 2) สแกนดูว่ามีอะไรรอบตัว (ESP32 เห็นเฉพาะ 2.4 GHz) ----
  Serial.println("[วินิจฉัย] กำลังสแกน WiFi รอบตัว (เห็นเฉพาะ 2.4 GHz)...");
  int n = WiFi.scanNetworks();
  if (n <= 0) {
    Serial.println("[วินิจฉัย] X ไม่เจอ WiFi เลยแม้แต่เครือข่ายเดียว");
    Serial.println("           -> รอบตัวเป็น 5GHz ทั้งหมด หรือเสาอากาศ/บอร์ดมีปัญหา");
    return;
  }
  Serial.printf("[วินิจฉัย] เจอ %d เครือข่าย:\n", n);
  bool found = false;
  for (int i = 0; i < n; i++) {
    String s = WiFi.SSID(i);
    Serial.printf("   %2d) %-28s %4d dBm %s\n", i + 1, s.c_str(), WiFi.RSSI(i),
                  WiFi.encryptionType(i) == WIFI_AUTH_OPEN ? "(เปิดโล่ง)" : "(มีรหัส)");
    if (saved.length() > 0 && s == saved) found = true;
  }

  if (saved.length() == 0) return;
  if (found) {
    Serial.println("[วินิจฉัย] OK เจอ \"" + saved + "\" ในลิสต์ -> ย่าน 2.4GHz ถูกต้อง");
    Serial.println("           ถ้าต่อไม่ได้ สาเหตุน่าจะเป็น \"รหัสผิด\"");
  } else {
    Serial.println("[วินิจฉัย] X ไม่เจอ \"" + saved + "\" ในลิสต์!");
    Serial.println("           -> เราเตอร์ปิด / อยู่ไกลเกิน / เป็น WiFi 5GHz (ESP32 รับไม่ได้)");
  }
}

void setup() {
  Serial.begin(115200);
  prefs.begin("flowlog", false);

  pinMode(FLOW_SENSOR_PIN, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(FLOW_SENSOR_PIN), onFlowPulse, FALLING);

  // 🚀 --- เริ่มระบบ WiFiManager ---
  WiFiManager wm;
  
  // ตั้งเวลารอต่อ Wi-Fi เดิมสูงสุด 15 วินาที (แก้ปัญหาค้าง waitForConnectResult)
  wm.setAPCallback(apCallback);   // พิมพ์บอกเมื่อเปิดหน้าตั้งค่า (เดิมเงียบ)
  wm.setDebugOutput(true);        // เปิด log ละเอียดของ WiFiManager
  wm.setConnectTimeout(15);

  // ปล่อยหน้า AP ให้ตั้งค่าเป็นเวลา 180 วินาที (3 นาที) ถ้าไม่มีใครตั้งค่า บอร์ดจะทำงานต่อหรือรีสตาร์ท
  wm.setConfigPortalTimeout(180);

  diagnoseWiFi();   // <<< บอกเหตุผลล่วงหน้าว่าทำไมอาจต่อไม่ได้

  Serial.println("\n[WiFi] กำลังเชื่อมต่อ Wi-Fi หรือสร้าง AP...");
  
  // ชื่อของ AP ที่จะปล่อยออกมา หากต่อ Wi-Fi ไม่สำเร็จ
  bool res = wm.autoConnect(AP_NAME, AP_PASSWORD);

  if(!res) {
    Serial.println("[WiFi] X ต่อไม่สำเร็จ และหมดเวลาหน้าตั้งค่า");
    Serial.println("[WiFi]   เหตุผล: " + wifiStatusText(WiFi.status()));
    Serial.println("[WiFi]   บอร์ดจะรีสตาร์ทแล้วลองใหม่ในอีก 3 วินาที...");
    delay(3000);
    ESP.restart();
  } 

  // เมื่อมาถึงตรงนี้ แสดงว่าเชื่อมต่อ Wi-Fi สำเร็จแล้ว
  Serial.println("\n[WiFi] เชื่อมต่อเครือข่ายสำเร็จ!");
  Serial.print("SSID     : "); Serial.println(WiFi.SSID());
  Serial.print("Signal   : "); Serial.print(WiFi.RSSI()); Serial.println(" dBm");
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());

  // --- เริ่มต้น Firebase และ NTP หลังจากต่อ Wi-Fi สำเร็จ ---
  config.host = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;
  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  configTime(GMT_OFFSET_SEC, 0, NTP_SERVER);

  Serial.println("\n=================================");
  Serial.println("Flow Sensor (ขาออก) ออนไลน์สำเร็จ!");
  Serial.println("=================================");

  loadTodayFromStorage();

  lastFlowCheckTime   = millis();
  lastPrefsSaveTime   = millis();
  lastHistoryPushTime = millis();
  lastWifiCheckTime   = millis();
}

// -------------------------------------------------------
void loop() {
  // ---- เฝ้าดู WiFi ทุก 10 วินาที ----
  // หากหลุด Wi-Fi ระหว่างทำงาน ให้พยายามเชื่อมต่อใหม่แบบไม่บล็อกการทำงาน (Non-blocking)
  // หน่วงเป็นรอบ 10 วิ เหมือนบอร์ดขาเข้า เพราะเดิมเรียก WiFi.reconnect() ทุกรอบลูป
  // (หลายร้อยครั้งต่อวินาที) ทำให้สแตกของ WiFi ถูกสั่งรีเซ็ตซ้ำจนต่อกลับไม่ติด
  if (millis() - lastWifiCheckTime >= WIFI_CHECK_INTERVAL) {
    lastWifiCheckTime = millis();
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println("[WiFi] หลุด - กำลังต่อใหม่...");
      WiFi.reconnect();
    }
  }

  // ---- คำนวณอัตราการไหลจริงจาก Flow Sensor ทุก 1 วินาที ----
  if (millis() - lastFlowCheckTime >= FLOW_CHECK_INTERVAL) {
    unsigned long elapsedMs = millis() - lastFlowCheckTime;

    noInterrupts();
    unsigned long pulses = flowPulseCount;
    flowPulseCount = 0;
    interrupts();

    float litersThisWindow = pulses / FLOW_CALIBRATION_FACTOR;
    currentFlowRate = litersThisWindow * (60000.0 / elapsedMs);
    flowStatus = (currentFlowRate > 0.5) ? "FLOW" : "NO FLOW"; 
    lastFlowCheckTime = millis();

    // ---- เช็กว่าข้ามชั่วโมง/วันใหม่หรือยัง ----
    String today   = getTodayDateKey();
    int    nowHour = getCurrentHour();

    if (nowHour != -1 && currentHour != -1 && nowHour != currentHour) {
      if (WiFi.status() == WL_CONNECTED) {
        String hourPath = "/water_system/history/" + currentDateKey +
                          "/hourly/" + hourKeyStr(currentHour) + "/used_liters";
        Firebase.setFloat(firebaseData, hourPath.c_str(), hourlyLiters);
      }
      Serial.printf("[Hour Rollover] ปิดยอดชั่วโมง %02d:00 = %.2f ลิตร\n", currentHour, hourlyLiters);
    }

    if (today != "" && today != currentDateKey) {
      currentDateKey = today;
      dailyLiters    = 0.0;
      saveTodayToStorage();
      Serial.printf("[Day Rollover] เริ่มนับยอดวันใหม่: %s\n", currentDateKey.c_str());
    }

    if (nowHour != -1 && nowHour != currentHour) {
      currentHour  = nowHour;
      hourlyLiters = 0.0;
      saveTodayToStorage();
    }

    // ---- สะสมยอดน้ำที่ใช้ไป ----
    if (litersThisWindow > 0 && currentDateKey != "") {
      dailyLiters += litersThisWindow;
      if (currentHour != -1) {
        hourlyLiters += litersThisWindow;
      }
    }

    Serial.printf("[Flow] %.2f L/min (%s) | วันนี้ใช้ไป %.2f ลิตร\n",
                  currentFlowRate, flowStatus.c_str(), dailyLiters);

    // ---- ส่งข้อมูลขึ้น Firebase ----
    if (WiFi.status() == WL_CONNECTED) {
      if (litersThisWindow > 0 && currentDateKey != "" &&
          millis() - lastHistoryPushTime >= HISTORY_PUSH_INTERVAL) {
        String basePath = "/water_system/history/" + currentDateKey;
        Firebase.setFloat(firebaseData, (basePath + "/used_liters").c_str(), dailyLiters);
        if (currentHour != -1) {
          String hourPath = basePath + "/hourly/" + hourKeyStr(currentHour) + "/used_liters";
          Firebase.setFloat(firebaseData, hourPath.c_str(), hourlyLiters);
        }
        lastHistoryPushTime = millis();
      }
    }
  }

  // ---- เซฟยอดสะสมลง flash ----
  if (millis() - lastPrefsSaveTime >= PREFS_SAVE_INTERVAL) {
    saveTodayToStorage();
    lastPrefsSaveTime = millis();
  }
}
